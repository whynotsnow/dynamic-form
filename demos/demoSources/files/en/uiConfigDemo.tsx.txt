import React from 'react';
import DynamicForm from '@/index';
import type { ComponentRegistryConfig, FormConfig, FormValues, UIConfig } from '@/exports';
import { Tag, Space, Form } from 'antd';
import type { FormInstance } from 'antd';
import { CheckCircleOutlined } from '@ant-design/icons';
import { useDemoInitHandlers } from './useDemoInitHandlers';
import { PriorityField, priorityEffect } from '../demos/customComponents/PriorityField';
import { useDemoSubmitResult } from './useDemoSubmitResult';
type DynamicUIFormValues = {
    title?: string;
    requester?: string;
    status?: 'open' | 'in_progress' | 'resolved';
    priority?: string;
    description?: string;
};
const getStatusUIEffect = (_changedValue: DynamicUIFormValues['status'], allValues: DynamicUIFormValues) => {
    const status = allValues.status;
    // Added: test dynamic UIConfig
    const uiConfigUpdates: Partial<UIConfig> = {};
    if (status === 'open') {
        uiConfigUpdates.formProps = {
            style: { backgroundColor: '#e6f4ff', border: '2px solid #1677ff' }
        };
        uiConfigUpdates.buttonProps = {
            type: 'primary',
            style: { backgroundColor: '#1677ff', borderColor: '#1677ff' }
        };
        uiConfigUpdates.cardProps = {
            style: { borderColor: '#1677ff', backgroundColor: '#e6f4ff' }
        };
    }
    else if (status === 'in_progress') {
        uiConfigUpdates.formProps = {
            style: { backgroundColor: '#fffbe6', border: '2px solid #faad14' }
        };
        uiConfigUpdates.buttonProps = {
            type: 'default',
            style: { backgroundColor: '#faad14', borderColor: '#faad14', color: 'white' }
        };
        uiConfigUpdates.cardProps = {
            style: { borderColor: '#faad14', backgroundColor: '#fffbe6' }
        };
    }
    else if (status === 'resolved') {
        uiConfigUpdates.formProps = {
            style: { backgroundColor: '#f6ffed', border: '2px solid #52c41a' }
        };
        uiConfigUpdates.buttonProps = {
            type: 'primary',
            style: { backgroundColor: '#52c41a', borderColor: '#52c41a' }
        };
        uiConfigUpdates.cardProps = {
            style: { borderColor: '#52c41a', backgroundColor: '#f6ffed' }
        };
    }
    return {
        formProps: uiConfigUpdates.formProps,
        buttonProps: uiConfigUpdates.buttonProps
        // This is not a group, so it has no Card component
        // cardProps: uiConfigUpdates.cardProps
    };
};
const getDescriptionUIEffect = (changedValue: DynamicUIFormValues['description']) => {
    const description = changedValue ?? '';
    return {
        componentProps: {
            rows: description.length > 2 ? 5 : 3,
            style: description.length > 2 ? { backgroundColor: '#fff2f0', border: '1px solid #ff4d4f' } : {}
        }
    };
};
const UIConfigDemo: React.FC<{ form?: FormInstance; hideSubmitButton?: boolean }> = ({
    form: externalForm,
    hideSubmitButton = false
}) => {
    const [localForm] = Form.useForm();
    const dynamicForm = externalForm ?? localForm;
    const { showSubmitResult, submitResultModal } = useDemoSubmitResult();
    // Dynamic label form config: demonstrate how useInitHandlers handles effect return values
    const dynamicFormConfig: FormConfig = {
        fields: [
            {
                id: 'title',
                component: 'TextInput',
                label: "Ticket title",
                required: true,
                formItemProps: {
                    label: (<Space>
              <span>🎫</span>
              <span style={{ fontWeight: 'bold', color: '#1890ff' }}>Ticket title</span>
              <Tag color="blue">Required</Tag>
            </Space>),
                    style: { borderLeft: '4px solid #1890ff', paddingLeft: '12px' },
                    className: 'required-field-highlight'
                }
            },
            {
                id: 'requester',
                component: 'TextInput',
                label: "Requester",
                required: true,
                formItemProps: {
                    label: (<Space>
              <span>👤</span>
              <span style={{ fontWeight: 'bold', color: '#52c41a' }}>Requester</span>
              <Tag color="green">Required</Tag>
            </Space>),
                    style: { borderLeft: '4px solid #52c41a', paddingLeft: '12px' },
                    className: 'required-field-highlight'
                }
            },
            {
                id: 'status',
                component: 'SelectField',
                label: "Ticket status",
                componentProps: {
                    options: [
                        { label: "Open", value: 'open' },
                        { label: "In progress", value: 'in_progress' },
                        { label: "Resolved", value: 'resolved' }
                    ]
                },
                formItemProps: {
                    label: (<Space>
              <CheckCircleOutlined style={{ color: '#52c41a' }}/>
              <span style={{ fontWeight: 'bold' }}>Ticket status</span>
              <Tag color="orange">Dynamic styling</Tag>
            </Space>),
                    style: { borderLeft: '4px solid #52c41a', paddingLeft: '12px' }
                },
                effect: (changedValue: DynamicUIFormValues['status'], allValues: DynamicUIFormValues) => getStatusUIEffect(changedValue, allValues)
            },
            {
                id: 'priority',
                component: 'PriorityField',
                label: "Ticket priority",
                componentProps: {
                    options: [
                        { label: "High", value: 'high' },
                        { label: "Medium", value: 'medium' },
                        { label: "Low", value: 'low' }
                    ]
                },
                formItemProps: {
                    label: (<Space>
              <span>🎯 Ticket priority</span>
              <Tag color="blue">Default</Tag>
            </Space>),
                    style: { borderLeft: '4px solid #faad14', paddingLeft: '12px' }
                },
                effect: priorityEffect
            },
            {
                id: 'description',
                component: 'TextArea',
                label: "Issue description",
                formItemProps: {
                    label: (<Space>
              <span>📝</span>
              <span style={{ fontWeight: 'bold', color: '#722ed1' }}>Details</span>
              <Tag color="purple">Optional</Tag>
            </Space>),
                    style: { borderLeft: '4px solid #722ed1', paddingLeft: '12px' }
                },
                componentProps: {
                    rows: 3
                },
                effect: (changedValue: DynamicUIFormValues['description']) => getDescriptionUIEffect(changedValue)
            }
        ]
    };
    // Dynamic label UI config
    const dynamicUIConfig: UIConfig = {
        colProps: {
            span: 8
        },
        formItemProps: {
            labelCol: { span: 24 },
            wrapperCol: { span: 24 }
        }
    };
    // Handle dynamic form submissions
    const handleDynamicSubmit = (values: FormValues) => {
        console.log('Dynamic form submitted::', values);
        showSubmitResult(values);
    };
    // Use useInitHandlers to handle effect return values
    useDemoInitHandlers();
    const componentRegistryConfig: ComponentRegistryConfig = {
        customComponents: { PriorityField }
    };
    return (<>
      <DynamicForm formConfig={dynamicFormConfig} uiConfig={dynamicUIConfig} componentRegistry={componentRegistryConfig} onSubmit={handleDynamicSubmit} form={dynamicForm}/>
      {submitResultModal}
    </>);
};
export default UIConfigDemo;
